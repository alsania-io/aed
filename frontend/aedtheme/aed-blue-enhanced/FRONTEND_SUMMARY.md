```

│   ├── index.html            # Main homepage with hero section

│   └── js/
│       ├── glass-ui.js       # UI controller and effects

└── aed-admin/
    ├── index.html            # Admin dashboard with analytics
    ├── css/style.css         # Admin-specific glass theme styles
    └── js/
        ├── glass-ui.js       # UI controller and admin effects
        ├── web3-provider.js  # Web3 integration for admin functions
        ├── contract-interface.js # Admin contract interactions

        └── main.js           # Main admin controller
```
